def format_srt_timestamp(seconds: float) -> str:
    """Converts seconds (float) into standard SRT timestamp format: HH:MM:SS,mmm"""
    total_ms = int(round(seconds * 1000))
    hours = total_ms // (3600 * 1000)
    remainder = total_ms % (3600 * 1000)
    minutes = remainder // (60 * 1000)
    remainder = remainder % (60 * 1000)
    secs = remainder // 1000
    ms = remainder % 1000
    return f"{hours:02d}:{minutes:02d}:{secs:02d},{ms:03d}"

def extract_audio_numpy(video_path: str, sampling_rate: int = 16000) -> np.ndarray:
    """
    Extracts audio from video directly into a 16kHz mono float32 numpy array using ffmpeg.
    Bypasses PyAV file opening bugs and disk I/O overhead.
    """
    if not os.path.isfile(video_path):
        raise FileNotFoundError(f"Video file not found: {video_path}")

    cmd = [
        "ffmpeg", "-nostdin", "-threads", "0",
        "-i", video_path,
        "-vn",
        "-f", "s16le",
        "-ac", "1",
        "-acodec", "pcm_s16le",
        "-ar", str(sampling_rate),
        "-"
    ]

    result = subprocess.run(cmd, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
    if result.returncode != 0:
        error_msg = result.stderr.decode("utf-8", errors="ignore")
        raise RuntimeError(f"ffmpeg audio extraction failed: {error_msg}")

    # Convert 16-bit signed PCM buffer to normalized float32 (-1.0 to 1.0)
    audio = np.frombuffer(result.stdout, dtype=np.int16).astype(np.float32) / 32768.0
    return audio

def chunk_into_reel_segments(
    segments_generator,
    max_words: int = 6,
    max_duration: float = 2.6,
    pause_threshold: float = 0.45
) -> List[Dict[str, Any]]:
    """
    Groups words from Whisper transcription into short, punchy 1.5 - 2.8s subtitle cards
    suitable for Reels, Shorts, and fast social media videos.
    """
    raw_chunks = []

    for seg in segments_generator:
        words = getattr(seg, "words", None)
        if words:
            current_words = []
            for w in words:
                if not current_words:
                    current_words.append(w)
                    continue

                chunk_dur = w.end - current_words[0].start
                word_gap = w.start - current_words[-1].end
                num_words = len(current_words)
                last_word = current_words[-1].word.strip()

                should_split = False
                if word_gap >= pause_threshold:
                    should_split = True
                elif num_words >= max_words:
                    should_split = True
                elif chunk_dur >= max_duration:
                    should_split = True
                elif last_word.endswith(('.', '?', '!', '।', ',')) and num_words >= 3:
                    should_split = True

                if should_split:
                    text = "".join(x.word for x in current_words).strip()
                    if text:
                        raw_chunks.append({
                            "start": current_words[0].start,
                            "end": current_words[-1].end,
                            "text": text
                        })
                    current_words = [w]
                else:
                    current_words.append(w)

            if current_words:
                text = "".join(x.word for x in current_words).strip()
                if text:
                    raw_chunks.append({
                        "start": current_words[0].start,
                        "end": current_words[-1].end,
                        "text": text
                    })
        else:
            seg_text = seg.text.strip()
            if seg_text:
                raw_chunks.append({
                    "start": seg.start,
                    "end": seg.end,
                    "text": seg_text
                })

    # 1. Post-process: merge tiny orphan fragments (< 0.75s or <= 1 word) into preceding card
    merged_chunks = []
    for c in raw_chunks:
        words = c["text"].strip().split()
        dur = c["end"] - c["start"]
        is_orphan = (len(words) <= 1 or dur < 0.75) and bool(merged_chunks)

        if is_orphan:
            merged_chunks[-1]["end"] = c["end"]
            merged_chunks[-1]["text"] = (merged_chunks[-1]["text"] + " " + c["text"]).strip()
        else:
            merged_chunks.append(dict(c))

    # 2. Cross-boundary split word stitching (e.g. 'reve' at end of card 1 and 'noo' at start of card 2)
    for i in range(len(merged_chunks) - 1):
        c1, c2 = merged_chunks[i], merged_chunks[i + 1]
        t1, t2 = c1["text"].rstrip(), c2["text"].lstrip()
        if re.search(r'(reve|रेवे)$', t1) and re.match(r'^(noo|nu|noog|नूग|नू|नु)', t2):
            c1["text"] = re.sub(r'(reve|रेवे)$', 'रेवेन्यू', c1["text"]).strip()
            c2["text"] = re.sub(r'^(noo|nu|noog|नूग|नू|नु)\w*\s*', '', c2["text"]).strip()

    return merged_chunks

def generate_subtitles(

    topic_hint: str = "",

# Script anchor: short, Devanagari, NO topic words. Keeps Whisper out of Urdu script.
SCRIPT_ANCHOR = "नमस्ते दोस्तों, यह वीडियो हिंदी और English मिलाकर बोला गया है।"

# Style template: natural Hinglish where English words sit in Latin script inside a
# Devanagari sentence. This teaches the *format* (English stays English) without
# biasing any domain. Keep generic words only.
STYLE_TEMPLATE = (
    "ये video बहुत interesting है। Basically एक simple idea ने पूरा game change कर दिया, "
    "और अब सब लोग उसे follow करते हैं।"
)


def hint_keywords(text: str, limit: int = 25) -> List[str]:
    """Pull candidate vocabulary out of a caption/title/hashtags the user pastes (best signal there is)."""
    if not text:
        return []
    text = text.replace("#", " ")
    seen, out = set(), []
    for tok in re.findall(r"[A-Za-z][A-Za-z0-9'&-]{2,}|[\u0900-\u097F]{3,}", text):
        k = tok.lower()
        if k not in seen:
            seen.add(k)
            out.append(tok)
    return out[:limit]


def build_prompt(topic_hint: str = "", extra_keywords: Iterable[str] = ()) -> Tuple[str, str]:
    """
    Returns (initial_prompt, hotwords).

    initial_prompt = script anchor + style template  (domain-neutral, always safe)
    hotwords       = caption/hint terms + discovered terms (domain-specific, optional)

    Why two channels: in faster-whisper, initial_prompt is a first-window prompt; with
    condition_on_previous_text=False it is dropped for later 30 s windows. `hotwords` is
    re-inserted for every window. Verify on your installed version (see notes in the reply).
    """
    kws: List[str] = []
    for k in list(hint_keywords(topic_hint)) + list(extra_keywords):
        if k.lower() not in {x.lower() for x in kws}:
            kws.append(k)
    return f"{SCRIPT_ANCHOR} {STYLE_TEMPLATE}", ", ".join(kws[:30])


def _supported_kwargs(model, **kw):
    """Drop kwargs this faster-whisper version doesn't know (e.g. older builds lack `hotwords`)."""
    params = inspect.signature(model.transcribe).parameters
    return {k: v for k, v in kw.items() if k in params}


def transcribe_words(model, audio, language: Optional[str] = "hi",
                     topic_hint: str = "", extra_keywords: Iterable[str] = (),
                     beam_size: int = 2) -> List[dict]:
    """Single transcription pass -> ONE flat word list across all segments (no per-segment reset)."""
    prompt, hot = build_prompt(topic_hint, extra_keywords)
    params = inspect.signature(model.transcribe).parameters
    if "hotwords" not in params and hot:
        # Fallback for older faster-whisper versions that lack hotwords
        prompt = f"{prompt} {hot}".strip()

    kw = _supported_kwargs(
        model,
        language=language,
        initial_prompt=prompt,
        hotwords=hot or None,
        task="transcribe",
        beam_size=beam_size,
        vad_filter=True,
        vad_parameters=dict(min_silence_duration_ms=400, speech_pad_ms=200, threshold=0.4),
        word_timestamps=True,
        condition_on_previous_text=False,
    )
    segments, info = model.transcribe(audio, **kw)
    words = []
    for seg in segments:
        for w in (getattr(seg, "words", None) or []):
            t = w.word.strip()
            if t:
                words.append({"start": float(w.start), "end": float(w.end), "text": t})
    return words


def discover_keywords(model, audio, seconds: float = 25.0, language: str = "hi") -> List[str]:
    """
    OPTIONAL cheap pass 1: greedy decode of the first ~25 s, harvest Latin-script tokens
    Whisper already got right. Helps *context*; it cannot rescue a brand Whisper has never
    heard (it will mishear it the same way). A user caption is a stronger signal.
    """
    clip = audio[: int(seconds * 16000)]
    kw = _supported_kwargs(
        model, language=language, initial_prompt=f"{SCRIPT_ANCHOR} {STYLE_TEMPLATE}",
        beam_size=1, vad_filter=True, without_timestamps=True,
        condition_on_previous_text=False,
    )
    segs, _ = model.transcribe(clip, **kw)
    text = " ".join(s.text for s in segs)
    freq: Dict[str, int] = {}
    for tok in re.findall(r"[A-Za-z][A-Za-z'-]{3,}", text):
        freq[tok] = freq.get(tok, 0) + 1
    return [t for t, _ in sorted(freq.items(), key=lambda kv: -kv[1])][:15]

def deva_to_phones(word: str) -> List[str]:
    """Rough Devanagari -> ARPAbet-like phones. Inherent schwa = 'AX' (cheap to skip later)."""
    s = unicodedata.normalize("NFD", word)
    out: List[str] = []
    pending = False
    i = 0
    while i < len(s):
        ch = s[i]
        nxt = s[i + 1] if i + 1 < len(s) else ""
        if ch in _CONS:
            if pending:
                out.append("AX")
            ph = _CONS[ch]
            if nxt == "\u093c":  # nukta
                ph = _NUKTA.get(ch, ph)
                i += 1
            out.append(ph)
            pending = True
        elif ch in _MATRA:
            pending = False
            out.extend(_MATRA[ch].split())
        elif ch == "\u094d":  # halant
            pending = False
        elif ch in _IND_VOWEL:
            if pending:
                out.append("AX")
                pending = False
            out.extend(_IND_VOWEL[ch].split())
        elif ch in ("\u0902", "\u0901"):  # anusvara / chandrabindu: nasal takes the place of the next consonant
            if pending:
                out.append("AX")
                pending = False
            out.append("M" if nxt and nxt in "पफबभम" else "NG" if nxt and nxt in "कखगघ" else "N")
        elif ch == "\u0903" or ch == "\u093c":
            pass
        i += 1
    if out[-2:] == ["NG", "G"]:  # word-final -ंग is just /ŋ/ (marketing, shopping)
        out = out[:-1]
    return out  # a trailing inherent schwa is never emitted (schwa deletion)


def _skeleton(phones: Sequence[str]) -> str:
    return "".join(_CLASS[p] for p in phones if p in _CLASS)


def _sub_cost(q: str, c: str) -> float:
    if q == c:
        return 0.0
    qv, cv = q in _VOWELS, c in _VOWELS
    if qv and cv:
        if "AX" in (q, c):
            return 0.15
        return 0.15 if any(q in s and c in s for s in _SOFT_VOWEL_SETS) else 0.45
    if qv != cv:
        return 1.5
    if {q, c} <= {"V", "W"}:
        return 0.0
    if {q, c} in ({"T", "TH"}, {"D", "DH"}):
        return 0.1
    if _CLASS.get(q) and _CLASS.get(q) == _CLASS.get(c):
        return 0.25
    return 1.0


def _indel_cost(p: str) -> float:
    if p == "AX":
        return 0.1
    if p in _VOWELS:
        return 0.45
    if p == "Y":
        return 0.3
    if p in ("N", "NG", "HH"):
        return 0.5
    return 0.9


def phone_distance(q: Sequence[str], c: Sequence[str]) -> float:
    """Weighted edit distance, normalised by the longer non-schwa length (0 = identical)."""
    n, m = len(q), len(c)
    prev = [0.0] * (m + 1)
    for j in range(1, m + 1):
        prev[j] = prev[j - 1] + _indel_cost(c[j - 1])
    for i in range(1, n + 1):
        cur = [prev[0] + _indel_cost(q[i - 1])] + [0.0] * m
        for j in range(1, m + 1):
            cur[j] = min(prev[j - 1] + _sub_cost(q[i - 1], c[j - 1]),
                         prev[j] + _indel_cost(q[i - 1]),
                         cur[j - 1] + _indel_cost(c[j - 1]))
        prev = cur
    denom = max(1, max(sum(p != "AX" for p in q), len(c)))
    return prev[m] / denom


def _expand_cmu(phones: Sequence[str]) -> List[str]:
    out: List[str] = []
    for p in phones:
        stress = re.sub(r"\D", "", p)
        p = re.sub(r"\d", "", p)
        if p == "AH" and stress == "0":
            p = "AX"
        if p == "ER":
            out.extend(["AX", "R"])
        else:
            out.append(p)
    return out


def load_cmudict_lexicon(min_zipf: float = 3.6) -> Dict[str, List[str]]:
    """English word -> phones. Filters to common words with wordfreq (if installed)."""
    import cmudict  # pip install cmudict

    try:
        from wordfreq import zipf_frequency
    except Exception:
        zipf_frequency = None
    lex: Dict[str, List[str]] = {}
    for w, prons in cmudict.dict().items():
        if not w.isalpha() or len(w) < 3:
            continue
        if zipf_frequency and zipf_frequency(w, "en") < min_zipf:
            continue
        lex[w] = _expand_cmu(prons[0])
    return lex


_HINDI_INFLECTION = re.compile(r"(ते|ता|ती|गा|गे|गी|ेगा|ेगी|रहा|रही|रहे|से|ने|को|का|के|की|में|पर)$")


def _has_english_signal(word: str) -> bool:
    """Spelling habits that almost only occur in Devanagari-written English."""
    if re.search(r"[ऑॉऍॅ]", word):
        return True
    if re.search(r"िंग$", word):
        return True
    if re.search(r"[\u095b\u095e]|ज़|फ़", word):
        return True
    return False


def hindi_protect_list(n: int = 10000, extra_file: str = "protected_extra.txt") -> Set[str]:
    """Top-n Hindi words (Devanagari) from wordfreq: real Hindi that must never be 'recovered' as English.
    Tokens with an English spelling signal (ऑ, -िंग, ज़/फ़) bypass this protection.
    Also loads extra words from protected_extra.txt if present."""
    words: Set[str] = set()
    try:
        from wordfreq import top_n_list
        words = set(top_n_list("hi", n))
    except Exception:
        pass

    for candidate in [extra_file, os.path.join(os.path.dirname(__file__), extra_file)]:
        if os.path.isfile(candidate):
            try:
                with open(candidate, "r", encoding="utf-8") as f:
                    for line in f:
                        line = line.strip()
                        if line and not line.startswith("#"):
                            words.add(line)
            except Exception:
                pass
            break
    return words


class LoanwordRecoverer:
    """
    lexicon  : {english_word: [phones]}  (see load_cmudict_lexicon)
    protected: set of Devanagari words that are real Hindi -> never converted
               (pass set(transliteration.HINDI_CONVERSATIONAL_MAP) | your own)
    curated  : Devanagari words your hand-written COMMON_LOANWORDS already handles -> always skipped,
               so the curated dictionary stays the first trust layer (pass set(COMMON_LOANWORDS))
    """

    def __init__(self, lexicon: Dict[str, List[str]], protected: Optional[Set[str]] = None,
                 curated: Optional[Set[str]] = None,
                 accept_with_signal: float = 0.34, accept_plain: float = 0.12, min_phones: int = 4):
        self.lex = lexicon
        self.protected = protected or set()
        self.curated = curated or set()
        self.t_sig, self.t_plain, self.min_phones = accept_with_signal, accept_plain, min_phones
        # SymSpell-style delete-1 index on the consonant skeleton => cheap candidate retrieval
        self.exact: Dict[str, List[str]] = {}
        self.dele: Dict[str, List[str]] = {}
        for w, ph in lexicon.items():
            sk = _skeleton(ph)
            if not sk:
                continue
            self.exact.setdefault(sk, []).append(w)
            for k in range(len(sk)):
                self.dele.setdefault(sk[:k] + sk[k + 1:], []).append(w)
        self._cache: Dict[str, Optional[str]] = {}
        self.decisions: List[Tuple[str, str]] = []  # (devanagari, english) -- review this to grow `protected`

    def _candidates(self, sk: str) -> Set[str]:
        cands: Set[str] = set(self.exact.get(sk, ()))
        cands.update(self.dele.get(sk, ()))                       # candidate has 1 extra consonant
        for k in range(len(sk)):                                   # query has 1 extra consonant
            cands.update(self.exact.get(sk[:k] + sk[k + 1:], ()))
        return cands

    def recover(self, deva_word: str) -> Optional[str]:
        """Return an English word for a Devanagari token, or None (leave it to the transliterator)."""
        if deva_word in self._cache:
            return self._cache[deva_word]
        res = self._recover(deva_word)
        self._cache[deva_word] = res
        return res

    def _recover(self, w: str) -> Optional[str]:
        core = re.sub(r"[^\u0900-\u097F]", "", w)
        signal = _has_english_signal(core)
        if not core or core in self.curated:
            return None
        if not signal and (core in self.protected or _HINDI_INFLECTION.search(core)):
            return None
        phones = deva_to_phones(core)
        if sum(p != "AX" for p in phones) < self.min_phones and not signal:
            return None
        sk = _skeleton(phones)
        if len(sk) < 2:
            return None
        best: List[Tuple[float, str]] = []
        for cand in self._candidates(sk):
            best.append((phone_distance(phones, self.lex[cand]), cand))
        if not best:
            return None
        best.sort()
        score, word = best[0]
        limit = self.t_sig if signal else self.t_plain
        if score > limit:
            return None
        if len(best) > 1 and best[1][0] - score < 0.02 and best[1][1] != word:
            # near-tie between different words: keep the shorter lexicon-closest one (already sorted);
            # ties are rare once the candidate set is skeleton-filtered.
            pass
        return word

    def merge_split_words(self, words: List[dict], max_gap: float = 0.15) -> List[dict]:
        """Heal 'रेवे'+'न्यू' style splits: join neighbours when the *joined* phones match an English word."""
        out: List[dict] = []
        i = 0
        while i < len(words):
            w = words[i]
            if i + 1 < len(words):
                n = words[i + 1]
                a = re.sub(r"[^\u0900-\u097F]", "", w["text"])
                b = re.sub(r"[^\u0900-\u097F]", "", n["text"])
                if a and b and n["start"] - w["end"] <= max_gap:
                    joined = deva_to_phones(a) + deva_to_phones(b)
                    sk = _skeleton(joined)
                    if len(sk) >= 3:
                        scored = sorted((phone_distance(joined, self.lex[c]), c) for c in self._candidates(sk))
                        if scored and scored[0][0] <= 0.15:
                            out.append({"start": w["start"], "end": n["end"], "text": scored[0][1]})
                            i += 2
                            continue
            out.append(w)
            i += 1
        return out

    def apply(self, words: List[dict]) -> List[dict]:
        """Full word-level pass: heal splits, then swap Devanagari loanwords for English spelling."""
        words = self.merge_split_words(words)
        res = []
        for w in words:
            if re.search(r"[\u0900-\u097F]", w["text"]):
                eng = self.recover(w["text"])
                if eng:
                    self.decisions.append((w["text"], eng))
                    record_decision("recovery", w["text"], eng, "cmu_phoneme_match",
                                    w.get("start", 0.0), w.get("end", 0.0))
                    tail = re.search(r"[.,!?।]+$", w["text"])
                    w = dict(w, text=eng + (tail.group(0) if tail else ""))
            res.append(w)
        return res


# --- brand / proper-noun snapping (words NOT in CMUdict, e.g. Dettol, Savlon) ---------
def _latin_skeleton(term: str) -> str:
    t = term.lower()
    for a, b in (("ph", "f"), ("th", "t"), ("sh", "s"), ("ch", "s"), ("ck", "k"), ("x", "ks"), ("qu", "kw")):
        t = t.replace(a, b)
    t = re.sub(r"c(?=[ei])", "s", t)
    t = t.replace("c", "k").replace("q", "k").replace("j", "s")
    m = {"b": "p", "f": "p", "v": "p", "w": "p", "p": "p", "d": "t", "t": "t", "g": "k", "k": "k", "z": "s", "s": "s",
         "m": "m", "n": "n", "l": "l", "r": "r", "y": "y", "h": "h"}
    return "".join(m[ch] for ch in t if ch in m)


def _edit1(a: str, b: str) -> bool:
    if abs(len(a) - len(b)) > 1:
        return False
    if len(a) == len(b):
        return sum(x != y for x, y in zip(a, b)) <= 1
    if len(a) > len(b):
        a, b = b, a
    return any(b[:i] + b[i + 1:] == a for i in range(len(b)))


def snap_to_terms(words: List[dict], terms: Iterable[str]) -> List[dict]:
    """
    Replace Devanagari tokens whose consonant skeleton is within 1 edit of a hint term
    (from the user's caption) with that term's exact spelling. 'दिटाल' -> 'Dettol'.
    Requires same first consonant class and >=3 consonants to avoid false hits.
    """
    index = [(t, _latin_skeleton(t)) for t in terms]
    index = [(t, s) for t, s in index if len(s) >= 3]
    res = []
    for w in words:
        core = re.sub(r"[^\u0900-\u097F]", "", w["text"])
        if core:
            sk = _skeleton(deva_to_phones(core))
            for t, ts in index:
                if len(sk) >= 3 and sk[0] == ts[0] and _edit1(sk, ts):
                    record_decision("brand_snap", w["text"], t, "caption_hint",
                                    w.get("start", 0.0), w.get("end", 0.0))
                    tail = re.search(r"[.,!?।]+$", w["text"])
                    w = dict(w, text=t + (tail.group(0) if tail else ""))
                    break
        res.append(w)
    return res


def _word_edit_dist(s1: str, s2: str) -> int:
    """Standard Levenshtein distance on characters."""
    m, n = len(s1), len(s2)
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    for i in range(m + 1):
        dp[i][0] = i
    for j in range(n + 1):
        dp[0][j] = j
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if s1[i - 1] == s2[j - 1]:
                dp[i][j] = dp[i - 1][j - 1]
            else:
                dp[i][j] = 1 + min(dp[i - 1][j], dp[i][j - 1], dp[i - 1][j - 1])
    return dp[m][n]


_COMMON_FUNCTION_WORDS = {
    "to", "do", "in", "on", "is", "us", "at", "it", "an", "as", "be", "by", "he", "if",
    "me", "my", "no", "or", "so", "up", "we", "the", "and", "for", "are", "but", "not",
    "you", "all", "any", "can", "had", "her", "was", "one", "our", "out", "day", "get",
    "has", "him", "his", "how", "man", "new", "now", "old", "see", "two", "way", "who",
    "boy", "did", "its", "let", "put", "say", "she", "too", "use", "with", "from", "that",
    "this", "have", "here", "were", "will", "your", "they", "them", "then", "than"
}


def unify_variants(words: List[dict],
                   min_freq: int = 1,
                   protected: Optional[Set[str]] = None,
                   lexicon: Optional[Dict[str, Any]] = None) -> List[dict]:
    """
    1. Clusters inconsistent Latin variants across the reel (e.g. 'Detol' / 'DTOL' / 'Dettol').
    2. Cross-script snaps Devanagari tokens to established Latin brands in the same reel.
    Guarded against short words and false positives.
    """
    if protected is None:
        protected = set(hindi_protect_list())
    try:
        from wordfreq import zipf_frequency
    except Exception:
        zipf_frequency = lambda w, lang: 0.0

    # Step 1: Collect Latin tokens
    latin_freq: Dict[str, int] = {}
    latin_words: Dict[str, str] = {}
    for w in words:
        clean = re.sub(r"^[^\w]+|[^\w]+$", "", w["text"])
        if clean.isalpha() and ord(clean[0]) < 128 and len(clean) >= 3:
            k = clean.lower()
            if k in _COMMON_FUNCTION_WORDS:
                continue
            latin_freq[k] = latin_freq.get(k, 0) + 1
            if k not in latin_words or (clean[0].isupper() and not latin_words[k][0].isupper()):
                latin_words[k] = clean

    # Step 2: Cluster Latin spelling variants
    clusters: List[Set[str]] = []
    keys = list(latin_freq.keys())
    for i in range(len(keys)):
        for j in range(i + 1, len(keys)):
            k1, k2 = keys[i], keys[j]
            if min(len(k1), len(k2)) >= 4 or (len(k1) >= 3 and len(k2) >= 3 and abs(len(k1) - len(k2)) <= 1):
                sk1 = _latin_skeleton(k1)
                sk2 = _latin_skeleton(k2)
                if sk1 and sk2 and (sk1 == sk2 or _edit1(sk1, sk2)):
                    max_d = 2 if max(len(k1), len(k2)) >= 6 else 1
                    if abs(len(k1) - len(k2)) <= max_d:
                        d = _word_edit_dist(k1, k2)
                        if d <= max_d:
                            merged = False
                            for c in clusters:
                                if k1 in c or k2 in c:
                                    c.add(k1)
                                    c.add(k2)
                                    merged = True
                                    break
                            if not merged:
                                clusters.append({k1, k2})

    # Pick canonical form for each cluster
    latin_map: Dict[str, str] = {}
    for c in clusters:
        def rank_candidate(k):
            orig = latin_words[k]
            freq = latin_freq[k]
            zf = zipf_frequency(k, "en")
            is_title = orig[0].isupper() and not orig.isupper()
            return (freq, zf, is_title, len(k))

        best_k = sorted(c, key=rank_candidate, reverse=True)[0]
        canonical = latin_words[best_k]
        for k in c:
            if k != best_k:
                latin_map[k] = canonical
                record_decision("unification", latin_words[k], canonical, f"cluster_canonical_freq={latin_freq[best_k]}")

    # Step 3: Build Latin brand index for cross-script snapping
    brand_index: List[Tuple[str, str]] = []
    for k, freq in sorted(latin_freq.items(), key=lambda kv: -kv[1]):
        zf = zipf_frequency(k, "en")
        # Only brands / proper nouns / rare terms (zipf < 3.8) are candidates for cross-script snapping
        if zf < 3.8:
            canonical = latin_map.get(k, latin_words[k])
            sk = _latin_skeleton(canonical)
            if len(sk) >= 3 and (canonical, sk) not in brand_index:
                brand_index.append((canonical, sk))

    # Step 4: Apply to all words
    out = []
    for w in words:
        t = w["text"]
        clean = re.sub(r"^[^\w]+|[^\w]+$", "", t)
        if clean.lower() in latin_map:
            canonical = latin_map[clean.lower()]
            tail = re.search(r"[.,!?।]+$", t)
            new_text = canonical + (tail.group(0) if tail else "")
            out.append(dict(w, text=new_text))
            continue

        core_deva = re.sub(r"[^\u0900-\u097F]", "", t)
        if core_deva and core_deva not in protected and not _HINDI_INFLECTION.search(core_deva):
            ph = deva_to_phones(core_deva)
            sk = _skeleton(ph)
            deva_ends_matra = bool(re.search(r"[\u093e-\u094c\u094e\u094f\u0955-\u0957]$", core_deva))
            deva_vowels = [p for p in ph if p in _VOWELS and p != "AX"]
            snapped = False
            for cand, csk in brand_index:
                cand_ends_vowel = cand.lower()[-1] in "aeiou"
                # Ending type must match: consonant-ending brand cannot match vowel-ending Hindi word (e.g. तितली ending in ी)
                if cand_ends_vowel != deva_ends_matra:
                    continue
                cand_vowels = [ch for ch in cand.lower() if ch in "aeiou"]
                if abs(len(cand_vowels) - len(deva_vowels)) > 1:
                    continue

                if len(sk) >= 3 and sk == csk:
                    tail = re.search(r"[.,!?।]+$", t)
                    new_text = cand + (tail.group(0) if tail else "")
                    record_decision("unification", t, new_text, f"cross_script_exact_skeleton={sk}",
                                    w.get("start", 0.0), w.get("end", 0.0))
                    out.append(dict(w, text=new_text))
                    snapped = True
                    break
            if not snapped:
                out.append(w)
        else:
            out.append(w)

    return out


def normalize_case(words: List[dict]) -> List[dict]:
    """
    Lowercase common words from CMUdict lexicon / English vocabulary unless:
    - It is at the start of a sentence (after '.', '?', '!', or index 0).
    - It is a cross-lingual named entity or regional food/proper noun (statistical rule: high Hindi cross-lingual frequency).
    - It is a low-frequency brand or proper noun (low English general frequency).
    No hardcoded word lists used.
    """
    try:
        from wordfreq import zipf_frequency
    except Exception:
        zipf_frequency = lambda w, lang: 0.0

    out = []
    for i, w in enumerate(words):
        t = w["text"]
        m = re.match(r"^([^\w]*)([\w]+)([^\w]*)$", t)
        if not m:
            out.append(w)
            continue
        pre, core, post = m.groups()
        if not core.isalpha() or ord(core[0]) >= 128:
            out.append(w)
            continue

        is_start = (i == 0) or bool(re.search(r"[.!?।]$", words[i - 1]["text"]))
        zf_en = zipf_frequency(core.lower(), "en")
        zf_hi = zipf_frequency(core.lower(), "hi")

        if is_start:
            norm = core.capitalize() if not core.isupper() else core
            reason = "sentence_start"
        elif zf_hi >= 2.5:
            # Cross-lingual proper noun / entity (e.g. India, Mumbai, Delhi)
            norm = core
            reason = f"cross_lingual_entity_hi_zipf={zf_hi:.2f}"
        elif zf_en >= 3.6:
            # Common English dictionary word occurring mid-sentence (e.g. talent, execution, monthly, revenue)
            norm = core.lower()
            reason = f"common_word_en_zipf={zf_en:.2f}"
        else:
            # Low-frequency brand or proper noun (e.g. Dettol, Savlon, Vada, Pav)
            norm = core
            reason = f"proper_or_brand_en_zipf={zf_en:.2f}"

        if norm != core:
            record_decision("casing", core, norm, reason, w.get("start", 0.0), w.get("end", 0.0))
            out.append(dict(w, text=pre + norm + post))
        else:
            out.append(w)
    return out

# 3. SEMANTIC REEL CHUNKING  (DP instead of greedy thresholds)
# =============================================================================
# Hindi is head-final with postpositions, so: never break BEFORE a postposition / auxiliary /
# light verb / clitic; avoid ENDING a card on a genitive or conjunction; prefer breaking
# BEFORE conjunctions and clause openers. Matching is on Whisper's Devanagari spelling
# (stable) plus common English function words.
_NO_BREAK_BEFORE = set("""का के की को से में पर तक ने है हैं था थी थे हो होगा होगी रहा रही रहे कर किया दिया लिया गया गई गए
हुआ हुई हुए ही भी तो वाला वाली वाले सकता सकती सकते चाहिए
लाख करोड़ हज़ार हजार प्लस परसेंट lakh crore thousand plus percent
of the to is are was were""".split())
_BAD_END = set("""का के की और या कि लेकिन पर क्योंकि जब अगर तो एक यह वह ये वो इस उस बहुत कुछ हर अपना अपनी अपने मेरा मेरी मेरे
the a an of in on to and or but with for my your our their this that""".split())
_GOOD_START = set("""और लेकिन पर कि क्योंकि जब अगर फिर या जो तभी इसलिए बस अब but and so because when if then""".split())
_NUM = re.compile(r"^[\d\u0966-\u096F][\d\u0966-\u096F.,]*$")
_PUNCT_END = re.compile(r"[.?!।]$")


def _norm(t: str) -> str:
    return re.sub(r"[^\w\u0900-\u097F]", "", t.lower())


def _break_cost(prev: dict, nxt: dict) -> float:
    p, n = _norm(prev["text"]), _norm(nxt["text"])
    gap = nxt["start"] - prev["end"]
    c = 0.0
    if _PUNCT_END.search(prev["text"]):
        c -= 3.0
    elif prev["text"].endswith(","):
        c -= 1.5
    if gap >= 0.35:
        c -= 3.0
    elif gap >= 0.2:
        c -= 1.2
    if n in _NO_BREAK_BEFORE:
        c += 4.0
    if p in _BAD_END:
        c += 3.0
    if n in _GOOD_START:
        c -= 1.5
    if _NUM.match(p):
        c += 3.0
    if re.match(r"^[a-z]+$", p) and re.match(r"^[a-z]+$", n) and p not in _BAD_END:
        c += 0.8  # "tissue|paper", "water|bottle": mild glue between consecutive English words
    return c


def semantic_chunk(words: List[dict], to_latin: Callable[[str], str] = lambda s: s,
                   min_dur: float = 1.3, target_dur: float = 2.0, max_dur: float = 2.8,
                   max_chars: int = 44, soft_max_words: int = 7,
                   pacing: str = "short") -> List[dict]:
    """
    words: [{'start','end','text'}] for the WHOLE clip (flatten Whisper segments first).
    to_latin: callable that returns the displayed (Roman) form, used only to measure line width.
    pacing: 'short' (reels/shorts: 1.0-2.8s, ~42 chars) or 'long' (youtube/16:9: 2.0-5.0s, ~76 chars).
    Returns cards [{'start','end','text'}].
    """
    if pacing == "long":
        min_dur = 2.0
        target_dur = 3.5
        max_dur = 5.0
        max_chars = 76
        soft_max_words = 14
    n = len(words)
    if n == 0:
        return []
    widths = [len(to_latin(w["text"])) + 1 for w in words]
    INF = float("inf")
    best = [INF] * (n + 1)
    back = [0] * (n + 1)
    best[0] = 0.0
    for j in range(1, n + 1):
        chars = 0
        window_limit = 24 if pacing == "long" else 14
        for i in range(j - 1, max(-1, j - window_limit), -1):
            chars += widths[i]
            dur = words[j - 1]["end"] - words[i]["start"]
            if dur > max_dur * 1.6 and i < j - 1:
                break
            nw = j - i
            cost = 0.0
            if dur > max_dur:
                cost += (100.0 + 200.0 * (dur - max_dur)) if nw > 1 else 20.0 * (dur - max_dur)
            if dur < min_dur and j != n:
                cost += 4.0 * (min_dur - dur)
            cost += 0.6 * (dur - target_dur) ** 2
            if nw > soft_max_words:
                cost += 2.0 * (nw - soft_max_words)
            if chars > max_chars:
                cost += 0.6 * (chars - max_chars)
            if nw == 1 and n > 1:
                cost += 3.0
            if i > 0:
                cost += _break_cost(words[i - 1], words[i])
            if best[i] + cost < best[j]:
                best[j] = best[i] + cost
                back[j] = i
    cards, j = [], n
    while j > 0:
        i = back[j]
        grp = words[i:j]
        cards.append({"start": grp[0]["start"], "end": grp[-1]["end"],
                      "text": " ".join(w["text"] for w in grp).strip()})
        j = i
    return cards[::-1]


def pad_display_times(cards: List[dict], min_dur: float = 1.0, hold: float = 0.3, gap: float = 0.04,
                      max_dur: float = 2.8, pacing: str = "short") -> List[dict]:
    """Hold each card a little after the last word and up to min_dur if room remains (never overlaps next)."""
    if pacing == "long":
        min_dur = 2.0
        hold = 0.45
        max_dur = 5.0
    out = []
    for k, c in enumerate(cards):
        limit = cards[k + 1]["start"] - gap if k + 1 < len(cards) else c["end"] + hold
        end = max(c["end"], min(c["end"] + hold, limit), min(c["start"] + min_dur, limit))
        end = min(end, max(c["end"], c["start"] + max_dur))  # padding never pushes a card past max_dur
        out.append(dict(c, end=max(c["end"], end)))
    return out


# =============================================================================
# GLUE: words -> final cards (what generate_subtitles() should call)
# =============================================================================
def words_to_cards(words: List[dict], to_hinglish: Callable[[str], str],
                   recoverer: Optional[LoanwordRecoverer] = None,
                   hint_terms: Iterable[str] = (),
                   reel_name: str = "reel",
                   pacing: str = "short") -> List[dict]:
    """
    words        : flat word list from transcribe_words()
    to_hinglish  : transliteration.convert_to_hinglish
    pacing       : 'short' (reels) or 'long' (youtube/16:9)
    Returns [{'start','end','original_text','hinglish_text'}]
    """
    if hint_terms:
        words = snap_to_terms(words, hint_terms)

    # 1. Unify inconsistent Latin variants and cross-script brand references
    words = unify_variants(words)

    # 2. Loanword recovery (Devanagari -> English)
    if recoverer is not None:
        words = recoverer.apply(words)

    # 3. Case normalization (eliminate stray capitals mid-sentence)
    words = normalize_case(words)

    # 4. Semantic chunking and padding
    cards = pad_display_times(semantic_chunk(words, to_latin=to_hinglish, pacing=pacing), pacing=pacing)

    # 5. Export decisions to TSV
    export_decisions(reel_name)

    return [{"start": c["start"], "end": c["end"], "original_text": c["text"],
             "hinglish_text": to_hinglish(c["text"])} for c in cards]

def _transliterate_hindi_word(word: str) -> str:
    """Converts a single Devanagari word into natural Latin Hinglish."""
    clean_devanagari = word.strip()
    if clean_devanagari in COMMON_LOANWORDS:
        return COMMON_LOANWORDS[clean_devanagari]

    # Aksharamukha ISO transliteration
    iso = transliterate.process('Devanagari', 'ISO', clean_devanagari)
    
    # In ISO transliteration, Devanagari 'च' is 'c' and 'छ' is 'ch'.
    # In English/Hinglish, 'ch' is used for 'च' and 'chh' for 'छ'.
    iso = re.sub(r'ch', 'chh', iso)
    iso = re.sub(r'(?<!c)c', 'ch', iso)
    
    # Map long vowels and specific letters
    iso = (
        iso.replace('ā', 'aa')
           .replace('ī', 'ee')
           .replace('ū', 'oo')
           .replace('ō', 'o')
           .replace('ē', 'e')
           .replace('ś', 'sh')
           .replace('ṣ', 'sh')
    )
    
    # Strip any remaining diacritics to clean ASCII
    ascii_str = unicodedata.normalize('NFKD', iso).encode('ASCII', 'ignore').decode('utf-8')
    
    # Clean up nasalized endings (anusvara)
    ascii_str = re.sub(r'em$', 'en', ascii_str)
    ascii_str = re.sub(r'aim$', 'ain', ascii_str)
    ascii_str = re.sub(r'om$', 'on', ascii_str)
    ascii_str = re.sub(r'im$', 'in', ascii_str)
    ascii_str = re.sub(r'um$', 'un', ascii_str)
    ascii_str = re.sub(r'm$', 'n', ascii_str)
    
    # Verbs ending in -naa or -na (e.g. seekhanaa -> seekhna, dekhanaa -> dekhna)
    ascii_str = re.sub(r'([bdfghjklmnpqrstvwxyz])a([bdfghjklmnpqrstvwxyz])naa?$', r'\1\2na', ascii_str)
    
    # Trailing schwa deletion for words ending in consonant + single 'a'
    if ascii_str.endswith('a') and len(ascii_str) > 2 and not ascii_str.endswith(('aa', 'ia', 'ya', 'ha')):
        ascii_str = ascii_str[:-1]

    # Check conversational map
    lower = ascii_str.lower()
    if lower in HINDI_CONVERSATIONAL_MAP:
        return HINDI_CONVERSATIONAL_MAP[lower]

    return ascii_str


def _transliterate_urdu_word(word: str) -> str:
    """Converts a single Perso-Arabic / Urdu word into natural Latin Hinglish."""
    clean_urdu = word.strip()
    if clean_urdu in URDU_WORD_MAP:
        return URDU_WORD_MAP[clean_urdu]

    # Fallback letter-by-letter mapping
    chars = [URDU_CHAR_MAP.get(c, c) for c in clean_urdu]
    res = ''.join(chars)
    
    # Clean vowel transitions
    res = re.sub(r'oo$', 'o', res)
    res = re.sub(r'ee$', 'i', res)
    return res

def convert_to_hinglish(text: str) -> str:
    """
    Takes any text (Devanagari Hindi, Perso-Arabic Urdu, English, or Code-mixed).
    Converts both Devanagari and Urdu portions into Latin script Hinglish,
    leaving English words, numbers, and symbols untouched.
    """
    if not text:
        return ""

    # Split while preserving delimiters (spaces, punctuation, symbols)
    tokens = re.split(r'(\s+|[.,!?:;\"\'\(\)\[\]{}<>\\/`~@#$%^&*+=_-]+)', text)
    result = []

    for token in tokens:
        # 1. Check if token contains Devanagari characters (\u0900 to \u097F)
        if re.search(r'[\u0900-\u097F]', token):
            result.append(_transliterate_hindi_word(token))
        # 2. Check if token contains Perso-Arabic / Urdu characters (\u0600 to \u06FF)
        elif re.search(r'[\u0600-\u06FF]', token):
            result.append(_transliterate_urdu_word(token))
        else:
            result.append(token)

    output = ''.join(result)

    # Post-process multi-word phrase patterns
    for pattern, replacement in PHRASE_NORMALIZATION:
        output = re.sub(pattern, replacement, output, flags=re.IGNORECASE)

    return output

function updateVideoOrientation() {
  const ext = currentFile ? currentFile.name.split('.').pop().toLowerCase() : "";
  const isAudio = (currentFile && currentFile.type.startsWith("audio/")) ||
                  ["mp3", "wav", "m4a", "aac", "flac", "ogg", "opus", "wma"].includes(ext);

  const w = videoPlayer.videoWidth;
  const h = videoPlayer.videoHeight;

  if (isAudio || !w || !h) {
    videoPlayer.classList.remove("video-portrait", "video-landscape", "video-square");
    videoPlayer.classList.add("media-audio");
    videoPlayer.style.aspectRatio = "";
    if (videoStage) videoStage.classList.add("is-audio");
    if (audioTrackBanner) audioTrackBanner.style.display = "flex";

    if (currentFile) {
      const sizeMB = (currentFile.size / (1024 * 1024)).toFixed(2);
      const dur = (videoPlayer.duration && !isNaN(videoPlayer.duration))
        ? ` • ${Math.round(videoPlayer.duration)}s`
        : "";
      fileMetaLabel.textContent = `${sizeMB} MB • Audio Track (${ext.toUpperCase() || "AUDIO"})${dur}`;
    }
    return;
  }

  // Video track detected
  if (audioTrackBanner) audioTrackBanner.style.display = "none";
  if (videoStage) videoStage.classList.remove("is-audio");
  videoPlayer.classList.remove("media-audio");

  videoPlayer.style.aspectRatio = `${w} / ${h}`;
  videoPlayer.classList.remove("video-portrait", "video-landscape", "video-square");

  if (h > w) {
    videoPlayer.classList.add("video-portrait");
  } else if (w === h) {
    videoPlayer.classList.add("video-square");
  } else {
    videoPlayer.classList.add("video-landscape");
  }

  // Format label update
  let formatName = "Landscape";
  const ratio = w / h;
  if (h > w) {
    if (ratio >= 0.54 && ratio <= 0.60) {
      formatName = "Vertical Reel (9:16)";
    } else if (ratio >= 0.75 && ratio <= 0.85) {
      formatName = "Portrait (4:5)";
    } else {
      formatName = `Vertical (${w}×${h})`;
    }
  } else if (w === h) {
    formatName = "Square (1:1)";
  } else {
    if (ratio >= 1.70 && ratio <= 1.82) {
      formatName = "Widescreen (16:9)";
    } else {
      formatName = `Landscape (${w}×${h})`;
    }
  }

  if (currentFile) {
    const sizeMB = (currentFile.size / (1024 * 1024)).toFixed(2);
    fileMetaLabel.textContent = `${sizeMB} MB • ${w}×${h} • ${formatName}`;
  }
}

  dropzone.style.display = "none";
  generateBtn.disabled = false;

  videoPlayer.className = "";

    videoPlayer.classList.add("media-audio");
    if (videoStage) videoStage.classList.add("is-audio");

  videoPlayer.src = videoUrl;
  videoPlayer.style.display = "block";

  if (videoPlayer.readyState >= 1) {
    updateVideoOrientation();
  }
}

function formatPlayerTime(seconds) {
  if (isNaN(seconds) || seconds === Infinity || seconds < 0) return "00:00";
  const totalSecs = Math.floor(seconds);
  const hrs = Math.floor(totalSecs / 3600);
  const mins = Math.floor((totalSecs % 3600) / 60);
  const secs = totalSecs % 60;
  const pad = n => String(n).padStart(2, "0");
  if (hrs > 0) {
    return `${pad(hrs)}:${pad(mins)}:${pad(secs)}`;
  }
  return `${pad(mins)}:${pad(secs)}`;
}

// --------------------------------------------------------------------------
// Real-Time Stepped Digital Sound Wave Equalizer (Web Audio API + Procedural Fallback)
// --------------------------------------------------------------------------
let audioCtx = null;
let audioAnalyser = null;
let audioSourceNode = null;
let visualizerAnimId = null;

function initAudioAnalyser() {
  if (audioCtx) return;
  try {
    const AudioContextClass = window.AudioContext || window.webkitAudioContext;
    if (!AudioContextClass) return;
    audioCtx = new AudioContextClass();
    audioAnalyser = audioCtx.createAnalyser();
    audioAnalyser.fftSize = 128;
    audioAnalyser.smoothingTimeConstant = 0.8;
    audioSourceNode = audioCtx.createMediaElementSource(videoPlayer);
    audioSourceNode.connect(audioAnalyser);
    audioAnalyser.connect(audioCtx.destination);
  } catch (err) {
    console.warn("Web Audio API binding note (using adaptive waveform):", err);
  }
}

function startVisualizerAnimation() {
  if (!audioVisualizerCanvas) return;
  if (!audioCtx) {
    initAudioAnalyser();
  }
  if (audioCtx && audioCtx.state === "suspended") {
    audioCtx.resume();
  }
  if (!visualizerAnimId) {
    renderAudioVisualizer();
  }
}

function renderAudioVisualizer() {
  if (!audioVisualizerCanvas) return;
  const ctx = audioVisualizerCanvas.getContext("2d");
  if (!ctx) return;

  const w = audioVisualizerCanvas.width;
  const h = audioVisualizerCanvas.height;
  ctx.clearRect(0, 0, w, h);

  const numCols = 32;
  const colWidth = 11;
  const colGap = 5;
  const totalWidth = numCols * colWidth + (numCols - 1) * colGap;
  const startX = Math.floor((w - totalWidth) / 2);

  const numBlocks = 9;
  const blockHeight = 5;
  const blockGap = 2;
  const totalBlockH = numBlocks * blockHeight + (numBlocks - 1) * blockGap;
  const startY = Math.floor((h - totalBlockH) / 2);

  let freqData = null;
  if (audioAnalyser && !videoPlayer.paused) {
    freqData = new Uint8Array(audioAnalyser.frequencyBinCount);
    audioAnalyser.getByteFrequencyData(freqData);
  }

  const isPlaying = !videoPlayer.paused && !videoPlayer.ended && videoPlayer.currentTime > 0;
  const now = performance.now() * 0.003;

  for (let c = 0; c < numCols; c++) {
    let amplitude = 0.12; // Resting baseline
    if (isPlaying) {
      if (freqData && freqData.length > 0) {
        const binIndex = Math.floor((c / numCols) * (freqData.length * 0.7));
        amplitude = freqData[binIndex] / 255;
      } else {
        const wave1 = Math.sin(now * 3.2 + c * 0.28);
        const wave2 = Math.cos(now * 5.4 - c * 0.45);
        const wave3 = Math.sin(now * 1.8 + c * 0.15);
        amplitude = Math.max(0.1, (wave1 * 0.35 + wave2 * 0.25 + wave3 * 0.2 + 0.45));
      }
    }

    const activeBlocks = Math.min(numBlocks, Math.max(1, Math.round(amplitude * numBlocks)));
    const x = startX + c * (colWidth + colGap);

    for (let b = 0; b < numBlocks; b++) {
      const y = startY + (numBlocks - 1 - b) * (blockHeight + blockGap);
      const isActive = b < activeBlocks;

      if (isActive) {
        if (b < 4) {
          ctx.fillStyle = "#facc15"; // Neon yellow register
        } else if (b < 7) {
          ctx.fillStyle = "#38bdf8"; // Cyan mid register
        } else {
          ctx.fillStyle = "#4ade80"; // Emerald green peak register
        }
      } else {
        ctx.fillStyle = "rgba(255, 255, 255, 0.07)"; // Unlit segment matrix
      }

      ctx.fillRect(x, y, colWidth, blockHeight);
    }
  }

  if (isPlaying) {
    visualizerAnimId = requestAnimationFrame(renderAudioVisualizer);
  } else {
    visualizerAnimId = null;
  }
}

  let progressInterval = setInterval(() => {
    elapsedSec++;
    let currentPct = parseInt(progressPctText.textContent) || 15;
    if (currentPct < 92) {
      currentPct = Math.min(92, currentPct + (currentPct < 60 ? 5 : (currentPct < 85 ? 2 : 1)));
      progressBarFill.style.width = `${currentPct}%`;
      progressPctText.textContent = `${currentPct}%`;
    }

    if (!isCached && elapsedSec < 12) {
      progressStatusText.textContent = `📥 Downloading Whisper [${selectedModel.toUpperCase()}] weights from HuggingFace...`;
    } else {
      const devName = selectedDevice.toUpperCase() === "AUTO" ? "GPU (CUDA)" : selectedDevice.toUpperCase();
      progressStatusText.textContent = `⚡ Neural transcription running on ${devName} (${elapsedSec}s elapsed)...`;
    }
  }, 1000);

  try {
    const res = await fetch("/api/transcribe", {
      method: "POST",
      body: formData
    });

    clearInterval(progressInterval);

    if (modelsStatusMap[selectedModel]) {
      modelsStatusMap[selectedModel].cached = true;
      updateModelStatusUI();
    }

    if (!res.ok) {
      const errData = await res.json();
      throw new Error(errData.detail || "Transcription failed");
    }

    const data = await res.json();
    currentResults = data;
    currentSegments = data.segments || [];

    // Complete Progress Bar

    progressPctText.textContent = "100%";
    progressStatusText.textContent = `Complete in ${data.elapsed_time.toFixed(1)}s (${data.speed_factor.toFixed(1)}x real-time)!`;

      progressPanel.style.display = "none";

    clearInterval(progressInterval);
    progressStatusText.textContent = `Error: ${err.message}`;
    progressBarFill.style.backgroundColor = "var(--pink)";
    generateBtn.disabled = false;
  }

function rebuildSrt(isHinglish = true) {
  if (!currentSegments || currentSegments.length === 0) return "";
  return currentSegments.map(seg => {
    const text = isHinglish ? seg.hinglish_text : seg.original_text;
    return `${seg.index}\n${seg.start_str} --> ${seg.end_str}\n${text}\n`;
  }).join("\n");
}

function triggerDownload(filename, content) {
  const blob = new Blob([content], { type: "text/plain;charset=utf-8" });
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url;
  a.download = filename;
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
  URL.revokeObjectURL(url);
}

downloadHinglishBtn.addEventListener("click", () => {
  const srtContent = rebuildSrt(true);
  const baseName = currentFile ? currentFile.name.replace(/\.[^/.]+$/, "") : "subtitles";
  triggerDownload(`${baseName}_hinglish.srt`, srtContent);
});

downloadOrigBtn.addEventListener("click", () => {
  const srtContent = rebuildSrt(false);
  const baseName = currentFile ? currentFile.name.replace(/\.[^/.]+$/, "") : "subtitles";
  triggerDownload(`${baseName}_original.srt`, srtContent);
});

downloadTxtBtn.addEventListener("click", () => {
  const textContent = currentSegments.map(s => s.hinglish_text).join("\n");
  const baseName = currentFile ? currentFile.name.replace(/\.[^/.]+$/, "") : "subtitles";
  triggerDownload(`${baseName}_hinglish.txt`, textContent);
});

function escapeHtml(text) {
  return String(text)
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#039;");
}

class TestLoanwordRecovery:
    @pytest.fixture(scope="class")
    def recoverer(self):
        lex = load_cmudict_lexicon(min_zipf=3.6)
        prot = set(HINDI_CONVERSATIONAL_MAP) | hindi_protect_list(n=10000)
        cur = set(COMMON_LOANWORDS)
        return LoanwordRecoverer(lex, protected=prot, curated=cur)

    def test_recovers_marketing_and_campaign(self, recoverer):
        # मार्केटिंग -> marketing
        res_m = recoverer.recover("मार्केटिंग")
        assert res_m == "marketing", f"Expected 'marketing', got {res_m}"

        # कैंपेन -> campaign
        res_c = recoverer.recover("कैंपेन")
        assert res_c == "campaign", f"Expected 'campaign', got {res_c}"

    def test_protects_hindi_words_from_false_recovery(self, recoverer):
        # कम must remain Hindi (never turn into English 'come')
        assert recoverer.recover("कम") is None

        # दूपते / डूबते has Hindi inflection -> must not match English
        assert recoverer.recover("दूपते") is None
        assert recoverer.recover("डूबते") is None

        # जलेगा has Hindi inflection -ेगा -> must not match English
        assert recoverer.recover("जलेगा") is None
        assert recoverer.recover("चलेगा") is None

        # फिरसे has postposition inflection -से -> must not match English 'fierce'
        assert recoverer.recover("फिरसे") is None

    def test_merges_split_loanwords(self, recoverer):
        # Heal 'रेवे' + 'न्यू' -> 'revenue'
        words = [
            {"start": 1.0, "end": 1.4, "text": "रेवे"},
            {"start": 1.45, "end": 1.8, "text": "न्यू"}
        ]
        healed = recoverer.merge_split_words(words)
        assert len(healed) == 1
        assert healed[0]["text"] == "revenue"

        # Devanagari tokens ڈیٹाल / डेटॉल should snap to the established Latin brand Detol

            {"start": 5.0, "end": 6.0, "text": "डेटॉल"},
            {"start": 10.0, "end": 11.0, "text": "डेतॉल"},
            {"start": 15.0, "end": 16.0, "text": "दिटाल"},
        ]
        unified = unify_variants(words)
        texts = [w["text"] for w in unified]
        assert texts == ["Detol", "Detol", "Detol", "Detol"]

            {"start": 4.0, "end": 5.0, "text": "to"},

# 5. SEMANTIC CHUNKER CONSTRAINTS
# =============================================================================
class TestSemanticChunker:
    def test_card_duration_ceiling(self):
        # Generate dummy stream of 50 words with realistic 0.3s duration each
        words = []
        t = 0.0
        for i in range(50):
            words.append({"start": round(t, 2), "end": round(t + 0.3, 2), "text": f"word{i}"})
            t += 0.35

        cards = pad_display_times(semantic_chunk(words, max_dur=2.8, pacing="short"), pacing="short")
        for c in cards:
            dur = c["end"] - c["start"]
            assert dur <= 2.81, f"Card duration exceeded 2.8s ceiling: {dur:.2f}s ({c})"
            assert dur >= 0.99, f"Card duration under min duration: {dur:.2f}s ({c})"

    def test_long_form_pacing_constraints(self):
        # 50 words in long-form mode should form longer cards (up to 5.0s, >= 2.0s)
        words = []
        t = 0.0
        for i in range(50):
            words.append({"start": round(t, 2), "end": round(t + 0.3, 2), "text": f"word{i}"})
            t += 0.35

        cards = pad_display_times(semantic_chunk(words, pacing="long"), pacing="long")
        # Fewer total cards than short-form
        short_cards = pad_display_times(semantic_chunk(words, pacing="short"), pacing="short")
        assert len(cards) < len(short_cards), f"Expected fewer cards in long-form ({len(cards)}) than short-form ({len(short_cards)})"
        for c in cards:
            dur = c["end"] - c["start"]
            assert dur <= 5.05, f"Long-form card exceeded 5.0s ceiling: {dur:.2f}s"
            assert dur >= 1.95, f"Long-form card under 2.0s min: {dur:.2f}s"

# 🎬 Hinglish Subtitle Generator (GPU Accelerated)

A high-performance tool that generates **Hinglish subtitles** (Hindi written in English alphabet / Romanized Hindi) for videos with **NVIDIA GPU Acceleration**.

Supports:
- **Mixed / Code-Switched Speech** (Hindi + English in the same video or sentence)
- **Pure Hindi Speech** (converted to natural Latin-script Hinglish)
- **Pure English Speech** (preserved as clean English subtitles)

---

## ⚡ Key Highlights
- **🚀 GPU Accelerated (CUDA)**: Up to **25x real-time speed** using NVIDIA RTX GPUs (Tensor Cores in `float16`).
- **100% Offline & Free**: Powered by local `faster-whisper` and `aksharamukha`. Zero cloud costs or API keys.
- **Zero Hallucination / No LLM Required**: Deterministic phonetic mapping guarantees speech-to-text timing and synchronization never drift.
- **Smart Loanword & Schwa Normalization**: Automatically normalizes tech loanwords (`video`, `subscribe`, `channel`, `comment`, `like`) and deletes Sanskrit-style trailing schwas (`hama` → `hum`, `bata` → `baat`).
- **Supports All Model Sizes**: `tiny`, `base`, `small`, `medium`, and **`large-v3`**.
- **Two Ways to Use**: Modern Streamlit Web UI or direct command-line interface.

---

### 1. Verification of Faster-Whisper Prompting vs Hotwords Theory (Task 1)
- **Source Inspection (`faster_whisper/transcribe.py`)**:
  - `initial_prompt` (lines 1143-1149) is appended to `all_tokens` prior to processing segment windows.
  - When `condition_on_previous_text=False`, line 1383 triggers `prompt_reset_since = len(all_tokens)` at the end of each window, causing `previous_tokens` to be empty for subsequent windows. As a result, `initial_prompt` is completely dropped in windows $k \ge 2$ ($t > 30\text{s}$).
  - `hotwords` (lines 1205, 1542-1548) is passed on every window iteration and injected directly into `prompt` right after `sot_prev`.
- **Empirical Proof on Reel Audio ($t > 30\text{s}$)**:
  - **No prompt**: Second-window terms output in Devanagari (`और इसी लाइन को डेटॉल ने पगड लिया`, `और मार्केटिंग, बिल्बोर्ड, एड, कैमपैन`, `डिटाल को अपने इंडस्ट्री का किंग`).
  - **initial_prompt only** (`billboard, campaign, Dettol, Savlon`): Window 1 recognized `Dettol` and `Savlon`, but Window 2 ($t > 30\text{s}$) **completely failed**, remaining Devanagari (`डेटॉल`, `मार्केटिंग, बिल्बोर्ड, एड, कैमपैन`, `डिटाल`).
  - **hotwords only** (`billboard, campaign, Dettol, Savlon`): Succeeded in **both** Window 1 and Window 2 (`और marketing, billboard, ad, campaign`, `Dettol ने पगड लिया`, `Dettol को अपने industry का king`).
- **Resolution**: Updated `build_prompt()` and `transcribe_words()` so domain terms and caption hints strictly route to `hotwords`, reserving `initial_prompt` for script and formatting templates (`SCRIPT_ANCHOR` + `STYLE_TEMPLATE`), with a fallback for older faster-whisper versions.

### 2. Lexical Variant Unification (Task 3)
- Implemented `unify_variants(words: List[dict])`:
  - Discovers Latin brand tokens and proper nouns across the reel (e.g. `Detol`, `Sablon`).
  - Clusters Latin spelling variants using consonant skeleton matching and character Levenshtein edit distance ($\le 1-2$).
  - Resolves cross-script Devanagari tokens (e.g. `दिटाल`, `डेटॉल`, `डेतॉल`) having exact consonant skeleton matches (`ttl`) to the reel's canonical Latin brand spelling (`Detol`).
  - Strict false-positive guards: protected common Hindi vocabulary (10,000 words + conversational map), inflection suffixes, and length $\ge 3$. Prevents Hindi words like `जलन` (`sln`) from snapping to `Sablon` (`spln`).

### 3. Sentence-Aware Case Normalization (Task 4)
- Implemented `normalize_case(words: List[dict])`:

  - Preserves capitalization for sentence starters (index 0 or following `.`, `?`, `!`, `।`), established proper nouns (`India`, `Mumbai`, `Vada Pav`), and low-frequency brand entities (`Dettol`, `Savlon`).

"""
review_decisions.py - Inspect lexical decisions (recovery, unification, casing)
and optionally add words to protected_extra.txt.
"""
import os
import sys
import glob
import argparse

sys.stdout.reconfigure(encoding="utf-8")


def list_log_files(log_dir: str = "logs"):
    if not os.path.exists(log_dir):
        return []
    files = glob.glob(os.path.join(log_dir, "decisions_*.tsv"))
    files.sort(key=lambda x: os.path.getmtime(x), reverse=True)
    return files


def read_tsv(filepath: str):
    rows = []
    with open(filepath, "r", encoding="utf-8", errors="ignore") as f:
        header = f.readline().strip().split("\t")
        for line in f:
            parts = line.strip().split("\t")
            if len(parts) >= 7:
                rows.append({
                    "timestamp": parts[0],
                    "stage": parts[1],
                    "start": float(parts[2]),
                    "end": float(parts[3]),
                    "original": parts[4],
                    "replaced": parts[5],
                    "reason": parts[6],
                })
    return rows


def print_decisions_table(rows, filename=""):
    print("=" * 80)
    print(f" DECISION LOG: {os.path.basename(filename) if filename else 'Combined'}")
    print("=" * 80)

    by_stage = {}
    for r in rows:
        by_stage.setdefault(r["stage"], []).append(r)

    print("\n[Summary by Stage]")
    for stg, items in by_stage.items():
        print(f"  - {stg:15s}: {len(items)} decisions")

    print("\n[Detailed Decisions]")
    print(f"{'TIME':12s} | {'STAGE':12s} | {'SPAN':13s} | {'ORIGINAL':18s} -> {'REPLACED':18s} | {'REASON'}")
    print("-" * 95)
    for r in rows:
        span = f"{r['start']:.2f}-{r['end']:.2f}s"
        print(f"{r['timestamp'][-8:]:12s} | {r['stage']:12s} | {span:13s} | {r['original']:18s} -> {r['replaced']:18s} | {r['reason']}")
    print("=" * 80)


def add_to_protected(word: str, target_file: str = "protected_extra.txt"):
    word = word.strip()
    if not word:
        return
    existing = set()
    if os.path.isfile(target_file):
        with open(target_file, "r", encoding="utf-8") as f:
            for line in f:
                line = line.strip()
                if line:
                    existing.add(line)
    if word in existing:
        print(f"'{word}' is already in {target_file}")
        return
    with open(target_file, "a", encoding="utf-8") as f:
        f.write(word + "\n")
    print(f"Added '{word}' to {target_file}")


def main():
    parser = argparse.ArgumentParser(description="Inspect subtitle generator lexical decisions.")
    parser.add_argument("--file", help="Path to specific decisions TSV file")
    parser.add_argument("--all", action="store_true", help="Inspect all log files combined")
    parser.add_argument("--protect", help="Append a Hindi word to protected_extra.txt")
    parser.add_argument("--list", action="store_true", help="List available decision logs")
    args = parser.parse_args()

    if args.protect:
        add_to_protected(args.protect)
        return

    log_files = list_log_files()

    if args.list:
        print(f"Found {len(log_files)} decision log(s):")
        for f in log_files:
            print("  ", f)
        return

    if not log_files and not args.file:
        print("No decision logs found in logs/ directory.")
        return

    target_files = []
    if args.file:
        target_files = [args.file]
    elif args.all:
        target_files = log_files
    else:
        target_files = [log_files[0]]

    for tf in target_files:
        rows = read_tsv(tf)
        print_decisions_table(rows, tf)


if __name__ == "__main__":
    main()

def get_gpu_memory_used_mb() -> float:
    try:
        res = subprocess.run(
            ["nvidia-smi", "--query-gpu=memory.used", "--format=csv,noheader,nounits"],

def compute_wer(reference_text: str, hypothesis_text: str) -> float:
    ref_words = [w.lower() for w in re.findall(r"\w+", reference_text)]
    hyp_words = [w.lower() for w in re.findall(r"\w+", hypothesis_text)]
    if not ref_words:
        return 0.0 if not hyp_words else 1.0
    r_len, h_len = len(ref_words), len(hyp_words)
    dp = [[0] * (h_len + 1) for _ in range(r_len + 1)]
    for i in range(r_len + 1):
        dp[i][0] = i
    for j in range(h_len + 1):
        dp[0][j] = j
    for i in range(1, r_len + 1):
        for j in range(1, h_len + 1):
            if ref_words[i - 1] == hyp_words[j - 1]:
                dp[i][j] = dp[i - 1][j - 1]
            else:
                dp[i][j] = 1 + min(dp[i - 1][j], dp[i][j - 1], dp[i - 1][j - 1])
    return (dp[r_len][h_len] / r_len) * 100.0
